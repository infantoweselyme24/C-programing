#include <stdio.h>

int main() {
    int arr[] = {10, 20, 5, 25, 15};
    int n = sizeof(arr) / sizeof(arr[0]);
    int first, second;
    if (n < 2) {
        printf("Array must contain at least two elements.\n");
        return 0;
    }
    first = second = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > first) {
            second = first;
            first = arr[i];
        } else if (arr[i] > second && arr[i] != first) {
            second = arr[i];
        }
    }
    if (first == second)
        printf("No distinct second largest number found.\n");
    else
        printf("Second largest number is: %d\n", second);
    return 0;
}

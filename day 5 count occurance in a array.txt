#include <stdio.h>
int main() {
    int arr[] = {1, 2, 2, 3, 4, 4, 4, 5};
    int n = sizeof(arr) / sizeof(arr[0]);
    int count[100] = {0};
    for (int i = 0; i < n; i++) {
        count[arr[i]]++;
    }
    printf("Element : Count\n");
    for (int i = 0; i < 100; i++) {
        if (count[i] > 0) {
            printf("%d : %d\n", i, count[i]);
        }
    }
    return 0;
}
